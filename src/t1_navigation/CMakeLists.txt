cmake_minimum_required(VERSION 3.16)
project(t1_navigation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(bondcpp REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(pluginlib REQUIRED)
add_library(exact_goal_lattice SHARED src/exact_goal_lattice.cpp)
ament_target_dependencies(exact_goal_lattice nav2_core nav2_smac_planner pluginlib)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS exact_goal_lattice LIBRARY DESTINATION lib)
add_executable(bounded_lifecycle_manager src/bounded_lifecycle_manager.cpp)
ament_target_dependencies(bounded_lifecycle_manager rclcpp lifecycle_msgs nav2_msgs std_srvs bondcpp)
install(TARGETS bounded_lifecycle_manager DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config maps behavior_trees DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(nav2_mppi_controller REQUIRED)
  find_package(yaml_cpp_vendor REQUIRED)
  find_package(yaml-cpp REQUIRED)
  ament_add_gtest(test_path_heading test/test_path_heading.cpp)
  ament_target_dependencies(test_path_heading nav2_mppi_controller)
  target_link_libraries(test_path_heading yaml-cpp)
  target_compile_definitions(test_path_heading PRIVATE
    T1_NAVIGATION_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/navigation.yaml")
endif()
ament_package()
